programa parcialturnop
procesos
  proceso juntarPapelesEsq(ES cantTotal:numero)
  comenzar
    Pos(3,20)
    repetir cantTotal
      tomarPapel
  fin
  proceso depositarPapeles(ES papeles:numero; E av:numero)
  comenzar
    BloquearEsquina(3,20)
    Pos(3,20)
    depositarPapel
    papeles:=papeles-1
    Pos(av,2)
    LiberarEsquina(3,20)
  fin
  proceso actualizarMin(E papeles:numero;E id:numero;ES idMin:numero; ES minp:numero)
  comenzar
    si(papeles < minp)
      minp:=papeles
      idMin:=id
  fin
  proceso enviarMin(E idMin:numero ;E cantTotal:numero)
  comenzar
    EnviarMensaje(idMin,r1)
    EnviarMensaje(cantTotal,r1)
    EnviarMensaje(idMin,r2)
    EnviarMensaje(cantTotal,r2)
    EnviarMensaje(idMin,r3)
    EnviarMensaje(cantTotal,r2)
  fin
  proceso juntarPapeles(ES papeles:numero)
  comenzar
    mientras (HayPapelEnLaEsquina)
      tomarPapel
      papeles:=papeles+1
  fin
  proceso realizarTramo(ES papeles:numero; E esq:numero)
  comenzar
    derecha
    repetir esq
      juntarPapeles(papeles)
      mover
    juntarPapeles(papeles)
  fin
  proceso enviarIDs
  comenzar
    EnviarMensaje(1,r1)
    EnviarMensaje(2,r2)
    EnviarMensaje(3,r3)
  fin
areas
  tramo1: AreaP (1,2,21,2)
  tramo2: AreaP(22,2,43,2)
  tramo3: AreaP(44,2,66,2)
  esquina: AreaPC (3,20,3,20)
  areajefe: AreaP (2,1,2,1)
robots
  robot recolector1
  variables
    papeles:numero
    miID:numero
    idMin:numero 
    cantTotal:numero
  comenzar
    papeles:=0
    RecibirMensaje(miID,rjefe)
    realizarTramo(papeles,20)
    EnviarMensaje(V,r2)
    EnviarMensaje(miID,rjefe)
    EnviarMensaje(papeles,rjefe)
    mientras(papeles>0)
      depositarPapeles(papeles,1) 
    RecibirMensaje(idMin,rjefe)
    RecibirMensaje(cantTotal,rjefe)
    si(idMin=miID)
      juntarPapelesEsq(cantTotal)
  fin
  robot recolector2
  variables
    papeles:numero
    miID:numero
    idMin:numero
    ok:boolean
    cantTotal:numero
  comenzar
    papeles:=0
    RecibirMensaje(miID,rjefe)
    RecibirMensaje(ok,r1)
    si (ok)
      realizarTramo(papeles,21)
    EnviarMensaje(V,r3)
    EnviarMensaje(miID,rjefe)
    EnviarMensaje(papeles,rjefe)
    mientras(papeles>0)
      depositarPapeles(papeles,22)
    RecibirMensaje(idMin,rjefe)
    RecibirMensaje(cantTotal,rjefe)
    si(idMin=miID)  
      juntarPapelesEsq(cantTotal)     
  fin
  robot recolector3
  variables
    papeles:numero
    miID:numero
    idMin:numero
    ok:boolean
    cantTotal:numero
  comenzar
    papeles:=0
    RecibirMensaje(miID,rjefe)
    RecibirMensaje(ok,r2)
    si(ok)
      realizarTramo(papeles,22)
    EnviarMensaje(miID,rjefe)
    EnviarMensaje(papeles,rjefe)
    mientras(papeles>0)
      depositarPapeles(papeles,44)
    EnviarMensaje(V,rjefe)
    
    RecibirMensaje(idMin,rjefe)
    RecibirMensaje(cantTotal,rjefe)
    si(idMin=miID)
      juntarPapelesEsq(cantTotal)
  fin
  robot robotjefe
  variables
    id:numero
    minp,idMin:numero
    papeles,cont:numero
    cantTotal:numero
    ok:boolean
  comenzar
    cantTotal:=0
    minp:=99
    idMin:=4
    cont:=0
    enviarIDs
    repetir 3
      RecibirMensaje(id,*)
      si (id=1)
        RecibirMensaje(papeles,r1)
        cantTotal:=cantTotal+papeles
      sino
        si(id=2)
          RecibirMensaje(papeles,r2)
          cantTotal:=cantTotal+papeles
        sino
          RecibirMensaje(papeles,r3)
          cantTotal:=cantTotal+papeles
      actualizarMin(papeles,id,idMin,minp)   
    RecibirMensaje(ok,r3)
    si(ok)
      enviarMin(idMin,cantTotal)
  fin
variables
  r1: recolector1
  r2: recolector2
  r3: recolector3
  rjefe:robotjefe 
comenzar
  AsignarArea(r1, tramo1)
  AsignarArea(r1,esquina)
  AsignarArea(r2, tramo2)
  AsignarArea(r2,esquina)
  AsignarArea(r3, tramo3)
  AsignarArea(r3,esquina)
  AsignarArea(rjefe,areajefe)
  Iniciar(r1, 1,2)
  Iniciar(r2, 22,2)
  Iniciar(r3, 44,2)
  Iniciar(rjefe,2,1)
fin
