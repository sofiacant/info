programa turnop
procesos
  proceso enviarId
  comenzar
    EnviarMensaje(1,r1)
    EnviarMensaje(2,r2)
    EnviarMensaje(3,r3)
  fin
  proceso juntarPapeles(ES papeles:numero)
  comenzar
    mientras HayPapelEnLaEsquina
      tomarPapel
      papeles:= papeles + 1 
  fin
  proceso recorrerCalle(ES tramo: numero; ES papeles:numero)
  comenzar
    derecha
    repetir tramo
      juntarPapeles(papeles)
      mover
    juntarPapeles(papeles)
  fin
  proceso deposito(ES papeles:numero)
  variables
    avI,caI:numero
  comenzar
    avI:=PosAv
    caI:=PosCa
    BloquearEsquina(3,20)
    Pos(3,20)
    repetir papeles
      si(papeles > 0)
        depositarPapel
        papeles:= papeles - 1
    Pos(avI,caI)
    LiberarEsquina(3,20)
  fin
  proceso robotMinimo(ES rMin:numero; ES id:numero; E min:numero; ES papeles:numero)
  comenzar
    si (id = 1)
      si (papeles < min)
        min:=papeles
        rMin:=1
      sino
        si (id = 2)
          si (papeles < min)
            min:=papeles
            rMin:=2
          sino
            si (papeles < min)
              min:=papeles
              rMin:=3
  fin
  proceso enviarIdMin(ES rMin:numero)
  comenzar
    EnviarMensaje(rMin,r1)
    EnviarMensaje(rMin,r2)
    EnviarMensaje(rMin,r3)
  fin
areas
  esquina: AreaPC (3,20,3,20)
  area1: AreaP (2,1,2,1)
  area2: AreaP (1,2,21,2)
  area3: AreaP (22,2,43,2)
  area4: AreaP (44,2,66,2)
robots
  robot recolector1
  variables
    id,idMin:numero
    tramo,papeles:numero
    ok:boolean
  comenzar
    papeles:=0
    tramo:=20
    RecibirMensaje(id,rj)
    recorrerCalle(tramo,papeles)
    deposito(papeles)
    EnviarMensaje(V,r2)
    EnviarMensaje(id,rj)
    EnviarMensaje(papeles,rj)
    RecibirMensaje(idMin,rj)
    si (idMin = id)
      Pos(3,20)
      juntarPapeles(papeles)
  fin
  robot recolector2
  variables
    id,idMin:numero
    tramo,papeles:numero
    ok:boolean
  comenzar
    papeles:=0
    tramo:=21
    RecibirMensaje(id,rj)
    RecibirMensaje(ok,r1)
    recorrerCalle(tramo,papeles)
    deposito(papeles)
    EnviarMensaje(V,r3)
    EnviarMensaje(id,rj)
    EnviarMensaje(papeles,rj)
    RecibirMensaje(idMin,rj)
    si (idMin = id)
      Pos(3,20)
      juntarPapeles(papeles)
  fin
  robot recolector3
  variables
    id,idMin:numero
    tramo,papeles:numero
    ok:boolean
  comenzar
    papeles:=0
    tramo:=22
    RecibirMensaje(id,rj)
    RecibirMensaje(ok,r2)
    recorrerCalle(tramo,papeles)
    deposito(papeles)
    EnviarMensaje(id,rj)
    EnviarMensaje(papeles,rj)
    RecibirMensaje(idMin,rj)
    si (idMin = id)
      Pos(3,20)
      juntarPapeles(papeles)
  fin
  robot jefe
  variables
    id,min,rMin:numero
    papeles:numero
  comenzar
    min:=999
    rMin:=4
    enviarId
    repetir 3
      RecibirMensaje(id,*)
      si (id = 1)
        RecibirMensaje(papeles,r1)
      sino
        si (id = 2)
          RecibirMensaje(papeles,r2)
        sino
          RecibirMensaje(papeles,r3)
      robotMinimo(rMin,id,min,papeles)   
    enviarIdMin(rMin)
  fin
variables
  r1: recolector1
  r2: recolector2
  r3: recolector3
  rj: jefe
comenzar
  AsignarArea(r1, area2)
  AsignarArea(r1, esquina)
  AsignarArea(r2, area3)
  AsignarArea(r2, esquina)
  AsignarArea(r3, area4)
  AsignarArea(r3, esquina)
  AsignarArea(rj, area1)
  Iniciar(r1, 1,2)
  Iniciar(r2, 22,2)
  Iniciar(r3, 44,2)
  Iniciar(rj, 2,1)
fin
